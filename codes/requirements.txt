# pip warns that torch pins 9.1.0.70; newer 9.x minors are backward compatible, so that's fine.

# IndicConformer. Install onnxruntime-gpu with --no-deps after everything else: faster-whisper
# pulls in the CPU "onnxruntime" package, which overwrites the GPU one if both are installed. Its model card says 1.20.1, but onnxruntime-gpu 1.20.1 was never published
# on PyPI; 1.20.2 is the matching patch release. Don't also install the CPU "onnxruntime" package.
onnx==1.20.1
onnxruntime-gpu==1.20.2
